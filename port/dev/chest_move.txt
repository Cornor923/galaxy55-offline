# cantina no modal (ent/Chest.hx del original): con la ventana abierta el juego sigue (se anda y se mira) y la ventana se
# cierra si el jugador se aleja mas de 0.5 de donde la abrio, o si la cantina se rompe.
# Esperado: abierta tras girar y tras moverse 0.3; cerrada tras 0.6 y tras andar; la madera (20) nunca se duplica ni se
# pierde: al cerrarse con una casilla cogida, al guardar con la ventana abierta (reload) y al romperla abierta (sale suelta).
wait 3
planet
wait 30
tp 130.5 24.5 top 0 -0.25
wait 20
setp 132 24 top BChest
give BWood 20
print inv
activate 132 24 36
wait 5
print chest
# con la ventana abierta no se salta (jetpack) ni se excava (Game.actionKeys = false): sigue en z 36, la cantina apuntada
# no acumula poder (0) y sigue ahi. lookat lleva la x ya envuelta (el planeta da la vuelta en 128: 132 = 4)
lookat 4.5 24.5 36.4
jet 1
key dig
wait 20
print select
jet 0
unkey
print pos
print block 132 24 36
print chest
# girar y mirar no la cierra
look 1.2 0.4
wait 10
print chest
shot chest_move_look
look 0 -0.25
wait 5
# pasar la madera a la cantina (casilla 9 del inventario -> casilla 0 de la cantina)
chestclick 1 9
chestclick 0 0
print chest
print inv
shot chest_move_open
# a 0.3 de donde se abrio: sigue abierta
nudge 0 0.3
wait 10
print pos
print chest
# coger la madera de la cantina y alejarse a mitad del cambio (0.6): se cierra sin mover nada
chestclick 0 0
nudge 0 0.3
wait 10
print pos
print chest
print inv
# se abre otra vez desde aqui y se aleja andando de lado
activate 132 24 36
wait 5
print chest
walk 0 1
wait 15
walk 0 0
wait 5
print pos
print chest
# guardar con la ventana abierta (Menu principal y Continuar): la madera vuelve al inventario antes de guardar
activate 132 24 36

chestclick 1 9
print chest

print inv
